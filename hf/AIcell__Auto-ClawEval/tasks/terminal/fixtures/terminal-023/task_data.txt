class HTMLSanitizer:
    """
    A basic HTML sanitizer that attempts to prevent XSS attacks.
    This implementation has several bugs that need to be identified and fixed.
    """

    ALLOWED_TAGS = {
        'p', 'br', 'strong', 'em', 'u', 'ol', 'ul', 'li',
        'h1', 'h2', 'h3', 'blockquote', 'code', 'pre'
    }

    ALLOWED_ATTRIBUTES = {
        'href': ['a'],
        'title': ['a', 'abbr'],
        'class': ['code', 'pre'],
    }

    DANGEROUS_PROTOCOLS = ['javascript:', 'vbscript:', 'data:']

    def __init__(self):
        self.errors = []

    def sanitize(self, input_html: str) -> str:
        """
        Main sanitization method.
        Removes dangerous HTML while preserving safe content.
        """
        if not input_html:
            return ""

        # Step 1: Remove script tags and content
        result = self._remove_script_tags(input_html)

        # Step 2: Remove event handlers
        result = self._remove_event_handlers(result)

        # Step 3: Remove dangerous protocols
        result = self._remove_dangerous_protocols(result)

        # Step 4: Strip disallowed tags
        result = self._strip_disallowed_tags(result)

        return result

    def _remove_script_tags(self, html_input: str) -> str:
        """Remove <script> tags and their content."""
        # BUG 1: This regex is case-sensitive and won't catch <SCRIPT> or <Script>
        pattern = r'<script[^>]*>.*?</script>'
        return re.sub(pattern, '', html_input, flags=re.DOTALL)

    def _remove_event_handlers(self, html_input: str) -> str:
        """Remove inline event handlers like onclick, onmouseover, etc."""
        # BUG 2: This pattern misses event handlers with spaces around '='
        # e.g., onclick = "alert(1)" would not be caught
        pattern = r'\s+on\w+="[^"]*"'
        return re.sub(pattern, '', html_input, flags=re.IGNORECASE)

    def _remove_dangerous_protocols(self, html_input: str) -> str:
        """Remove dangerous protocol handlers from href attributes."""
        result = html_input
        for protocol in self.DANGEROUS_PROTOCOLS:
            # BUG 3: This doesn't handle URL-encoded versions like javascript&#58;
            # or whitespace injection like java\nscript:
            pattern = f'href=["\']\\s*{re.escape(protocol)}[^"\']*["\']'
            result = re.sub(pattern, 'href="#"', result, flags=re.IGNORECASE)
        return result

    def _strip_disallowed_tags(self, html_input: str) -> str:
        """Strip HTML tags that are not in the allowed list."""
        def replace_tag(match):
            tag_content = match.group(1)
            tag_name = tag_content.split()[0].lower().lstrip('/')
            if tag_name in self.ALLOWED_TAGS:
                return match.group(0)
            # BUG 4: Returns empty string but should escape the tag content
            # to prevent tag reconstruction attacks
            return ''

        pattern = r'<([^>]+)>'
        return re.sub(pattern, replace_tag, html_input)

    def escape_output(self, text: str) -> str:
        """Escape special HTML characters for safe output."""
        return html.escape(text, quote=True)
